cmake_minimum_required(VERSION 3.18)
project(cuda_gpu_fundamentals CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)

# Real CUDA targets are only built when a CUDA toolchain is present (not on macOS).
include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES native)
  endif()
  message(STATUS "CUDA found: building kernel.cu targets")
else()
  message(STATUS "CUDA not found: building CPU simulators only")
endif()

file(GLOB_RECURSE SIM_SOURCES "${CMAKE_SOURCE_DIR}/[0-9][0-9]_*/*/sim.cpp")
foreach(src ${SIM_SOURCES})
  get_filename_component(dir ${src} DIRECTORY)
  get_filename_component(name ${dir} NAME)
  add_executable(sim_${name} ${src})
  target_include_directories(sim_${name} PRIVATE ${CMAKE_SOURCE_DIR}/common/sim)
  target_link_libraries(sim_${name} PRIVATE Threads::Threads)
  target_compile_options(sim_${name} PRIVATE -Wall -Wextra)

  if(CMAKE_CUDA_COMPILER AND EXISTS ${dir}/kernel.cu)
    add_executable(cuda_${name} ${dir}/kernel.cu)
    set_target_properties(cuda_${name} PROPERTIES CUDA_STANDARD 17)
  endif()
endforeach()
